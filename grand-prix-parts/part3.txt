function placeCar(o, x, z, a, speed, steer, braking, dt, mine) {
  o.root.visible = true;
  o.root.position.set(x, 0, z); o.root.rotation.y = -a;
  o.spin += speed * dt / 0.36;
  if (o.pitT > 0) {                                     // pneus qui clignotent le temps du changement
    o.pitT -= dt;
    const on = Math.floor(o.pitT * 10) % 2 === 0;
    for (const w of o.wheels) w.spin.visible = on;
  } else if (o.wheels.length && !o.wheels[0].spin.visible) {
    for (const w of o.wheels) w.spin.visible = true;
  }
  for (const w of o.wheels) { w.spin.rotation.z = -o.spin; w.steer.rotation.y = w.front ? -steer * 0.4 : 0; }
  const ax = (speed - o.lastSpeed) / Math.max(dt, 0.001); o.lastSpeed = speed;
  const k = Math.min(1, dt * 8);
  o.pitch += (clamp(ax * 0.0009, -0.05, 0.03) - o.pitch) * k;
  o.roll += (-steer * Math.min(1, Math.abs(speed) / 60) * 0.035 - o.roll) * k;
  o.body.rotation.z = o.pitch; o.body.rotation.x = o.roll;
  o.brake.color.setHex(braking ? 0xff2a1a : 0x550000);
  if (o.brakeGlow) o.brakeGlow.opacity += ((braking ? 0.95 : 0) - o.brakeGlow.opacity) * Math.min(1, dt * 20);
  if (o.flame) o.flame.opacity = mine ? Math.min(1, fxPop) : 0;
}

function updateCamera(dt, t) {
  const racing = phase === "countdown" || phase === "race" || phase === "finished";
  let fov = 55;
  if (!racing) {
    const s = slotPose(3), ang = t * 0.00016, rad = 32;
    camera.position.set(s.x + Math.cos(ang) * rad, 9, s.z + Math.sin(ang) * rad);
    camera.lookAt(s.x, 1.2, s.z);
  } else {
    camS.a += wrap(me.a - camS.a) * (1 - Math.exp(-(camS.mode === 2 ? 40 : 6) * dt));
    const v = Math.abs(me.speed), ca = Math.cos(camS.a), sa = Math.sin(camS.a);
    let dist, hgt, look, lookY;
    if (camS.mode === 0) { dist = 8.5 + v * 0.02; hgt = 3.2 + v * 0.004; look = 14; lookY = 1.3; fov = 66 + v * 0.16; }
    else if (camS.mode === 1) { dist = 14; hgt = 7.5; look = 12; lookY = 1.0; fov = 62 + v * 0.10; }
    else { dist = -0.3; hgt = 1.2; look = 22; lookY = 1.0; fov = 78 + v * 0.10; }
    camS.shake *= Math.exp(-4 * dt);
    const amp = camS.shake * 0.5 + (me.dist > OFFD ? 0.07 * Math.min(1, v / 40) : 0);
    const jx = (Math.random() - 0.5) * amp, jy = (Math.random() - 0.5) * amp, jz = (Math.random() - 0.5) * amp;
    camera.position.set(me.x - ca * dist + jx, hgt + jy, me.z - sa * dist + jz);
    camera.lookAt(me.x + ca * look, lookY, me.z + sa * look);
    camS.roll += (-me.steer * Math.min(1, v / 60) * 0.035 - camS.roll) * Math.min(1, dt * 5);
    camera.rotateZ(camS.roll);
  }
  if (Math.abs(camera.fov - fov) > 0.05) { camera.fov += (fov - camera.fov) * Math.min(1, dt * 6); camera.updateProjectionMatrix(); }
  sky.position.copy(camera.position);
  if (sunSprite && sunDir) sunSprite.position.copy(camera.position).addScaledVector(sunDir, 3000);
}

      placeCar(o2, r.rx, r.rz, r.ra, r.v || 0, r.sv || 0, false, dt);

    renderer.shadowMap.enabled = true;
    renderer.shadowMap.type = THREE.PCFSoftShadowMap;
    if ("outputColorSpace" in renderer) renderer.outputColorSpace = THREE.SRGBColorSpace;
    if ("toneMapping" in renderer) { renderer.toneMapping = THREE.ACESFilmicToneMapping; renderer.toneMappingExposure = 1.05; }
  } catch (e) { GL_ERR = "Ton navigateur ne peut pas afficher la 3D (WebGL). Essaie Chrome ou Firefox."; renderer = null; return false; }

const GEAR_TOP = [24, 40, 54, 66, 76, 84, 91, 100];
function gearInfo(v) {
  let g = 0; while (g < 7 && v > GEAR_TOP[g]) g++;
  const lo = g ? GEAR_TOP[g - 1] : 0, hi = GEAR_TOP[g];
  return { gear: g + 1, rpm: v < 0.5 ? 0.2 : 0.32 + 0.68 * clamp((v - lo) / (hi - lo), 0, 1) };
}

function noiseBuf(ac, sec, brown) {
  const n = Math.floor(ac.sampleRate * sec), b = ac.createBuffer(1, n, ac.sampleRate), d = b.getChannelData(0);
  let last = 0;
  for (let i = 0; i < n; i++) {
    const w = Math.random() * 2 - 1;
    if (brown) { last = (last + 0.02 * w) / 1.02; d[i] = last * 3.5; } else d[i] = w;
  }
  return b;
}
function impulseBuf(ac, sec, decay) {
  const n = Math.floor(ac.sampleRate * sec), b = ac.createBuffer(2, n, ac.sampleRate);
  for (let ch = 0; ch < 2; ch++) {
    const d = b.getChannelData(ch);
    for (let i = 0; i < n; i++) d[i] = (Math.random() * 2 - 1) * Math.pow(1 - i / n, decay);
  }
  return b;
}
function satCurve(k) {
  const n = 1024, c = new Float32Array(n);
  for (let i = 0; i < n; i++) { const x = i * 2 / (n - 1) - 1; c[i] = Math.tanh(x * k); }
  return c;
}

function updateAudio(dt, t) {
  const A = audio; if (!A) return;
  const ac = A.ac, now = ac.currentTime;
  const racing = phase === "race" || phase === "countdown" || phase === "finished";
  A.master.gain.setTargetAtTime(muted ? 0 : 0.9, now, 0.05);
  const E = A.eng;
  if (!racing) {
    E.bus.gain.setTargetAtTime(0, now, 0.05);
    ["tire", "gravel", "wind", "crowd"].forEach(k => A[k].g.gain.setTargetAtTime(0, now, 0.08));
    A.kerb.g.gain.setTargetAtTime(0, now, 0.05);
    A.voices.forEach(vc => vc.g.gain.setTargetAtTime(0, now, 0.05));
    A.prevPhase = phase; A.lit = 0; A.cheer = 0; return;
  }
  const v = Math.abs(me.speed), thr = me.thr || 0, gi = gearInfo(v);

  // Feux de départ : clic mécanique par feu, sans tonalité électronique
  if (phase === "countdown") {
    const lit = clamp(Math.floor((t - cdStart) / 600), 0, 5);
    if (lit > A.lit) { A.lit = lit; sfxBeep(false); }
  } else if (A.prevPhase === "countdown" && phase === "race") { sfxBeep(true); A.cheer = Math.max(A.cheer, 0.5); }
  A.prevPhase = phase;

  // Régime moteur : il monte vite, il retombe plus lentement
  let tgt = gi.rpm, eThr = thr;
  if (phase === "countdown") { tgt = 0.38 + A.lit * 0.085 + 0.025 * Math.sin(t * 0.045); eThr = 0.55; }
  else if (v < 20 && thr > 0) tgt = Math.max(tgt, 0.5 + 0.3 * thr);
  A.rpm += (tgt - A.rpm) * (1 - Math.exp(-dt * (tgt > A.rpm ? 10 : 5.5)));
  const rpm = clamp(A.rpm, 0, 1.02);

  // Changements de rapport : coupure, "clac", et pétarade en rétrogradant
  if (phase === "race" && gi.gear !== A.gear && v > 6) {
    const upsh = gi.gear > A.gear;
    A.dip = upsh ? 1 : 0.55; sfxClunk(upsh); if (!upsh) sfxPop(0.6);
  }
  A.gear = gi.gear; A.dip *= Math.exp(-dt * 16);

  // Moteur : trois couches synchronisées — combustion, échappement et admission/turbo.
  const engineRPM=820+rpm*10180;
  const fireHz=Math.max(20,(engineRPM/60)*3);
  const rough=1+0.006*Math.sin(t*1.7)+0.003*Math.sin(t*4.2);
  const f=fireHz*rough;
  E.o1.frequency.setTargetAtTime(f,now,.018);
  E.o2.frequency.setTargetAtTime(f*1.003,now,.018);
  E.o3.frequency.setTargetAtTime(f*2.0,now,.022);
  E.o4.frequency.setTargetAtTime(f*4.01,now,.028);
  E.o5.frequency.setTargetAtTime(1200+rpm*7200,now,.035);
  E.o2.detune.setTargetAtTime(2.5+2*Math.sin(t*.31),now,.08);
  const lift=phase==="race"&&thr<.05;
  const load=.28+eThr*1.65;
  E.lp.frequency.setTargetAtTime(1150+rpm*4800+eThr*2600-(lift?500:0),now,.045);
  E.g3.gain.setTargetAtTime(.028+.15*rpm*load,now,.055);
  E.g4.gain.setTargetAtTime(.002+.035*rpm*rpm*(.25+.75*eThr),now,.055);
  E.g5.gain.setTargetAtTime(.012+.035*eThr+.018*rpm,now,.06);
  E.exF.frequency.setTargetAtTime(260+rpm*620+eThr*260,now,.06);
  E.exG.gain.setTargetAtTime(.015+.095*rpm*(.35+.65*eThr),now,.06);
  E.gb.frequency.setTargetAtTime(1700+rpm*5200+eThr*1300,now,.05);
  E.gg.gain.setTargetAtTime(.002+.032*eThr*(.25+rpm),now,.055);
  const cam=camS.mode===2?1.12:camS.mode===1?.84:1;
  const engineVol=(.095+.105*eThr+.055*rpm)*cam;
  E.bus.gain.setTargetAtTime(
    engineVol * (1 - 0.62 * A.dip) * (phase === "finished" ? 0.30 : 1),
    now, 0.035
  );
  // Pétarade quand on lâche l'accélérateur à haut régime
  if (lift && rpm > 0.55 && v > 35 && Math.random() < dt * 9 * (rpm - 0.45)) sfxPop(0.6 + Math.random() * 0.4);

  // Pneus qui crissent
  const slip = phase === "race" ? slipOf(me) : 0;
  A.tire.g.gain.setTargetAtTime(slip * 0.17, now, 0.04);
  A.tire.fl.frequency.setTargetAtTime(1250 + 700 * slip + 120 * Math.sin(t * 0.03), now, 0.05);
  // Vibreurs et gravier
  const kerb = phase === "race" && me.dist > TRACK_W / 2 - 1 && me.dist <= OFFD && v > 5;
  A.kerb.g.gain.setTargetAtTime(kerb ? clamp(v / 60, 0.2, 1) * 0.55 : 0, now, 0.03);
  A.kerb.lfo.frequency.setTargetAtTime(clamp(v / 3.0, 4, 45), now, 0.05);
  A.gravel.g.gain.setTargetAtTime(me.dist > OFFD && v > 3 ? clamp(v / 50, 0, 1) * 0.3 : 0, now, 0.05);
  // Vent
  A.wind.fl.frequency.setTargetAtTime(400 + v * 9, now, 0.1);
  A.wind.g.gain.setTargetAtTime(Math.pow(v / 100, 1.6) * 0.12 * (camS.mode === 2 ? 1.4 : 1), now, 0.1);
  // Foule : on l'entend en passant devant les tribunes, et elle explose à l'arrivée
  let dmin = 1e9;
  for (const i of [6, 30, 60]) dmin = Math.min(dmin, Math.hypot(me.x - TX[i], me.z - TZ[i]));
  const near = Math.pow(clamp(1 - (dmin - 30) / 240, 0, 1), 2);
  A.cheer = Math.max(0, A.cheer - dt * 0.3);
  A.crowd.g.gain.setTargetAtTime(0.012 * near + 0.16 * A.cheer * (0.4 + 0.6 * near), now, 0.2);

  // Les autres voitures : volume selon la distance, gauche/droite, hauteur du son (Doppler)
  const list = [];
  const add = (x, z, a, sp) => list.push({ x, z, vx: Math.cos(a) * sp, vz: Math.sin(a) * sp, sp: Math.abs(sp), d: Math.hypot(x - me.x, z - me.z) });
  for (const id in remotes) { const r = remotes[id]; if (id[0] !== "b" && !people[id]) continue; add(r.rx, r.rz, r.ra, r.v || 0); }
  for (const b of bots) add(b.x, b.z, b.a, b.speed);
  list.sort((p, q) => p.d - q.d);
  const sm = Math.sin(me.a), cm = Math.cos(me.a);
  for (let i = 0; i < A.voices.length; i++) {
    const vc = A.voices[i], o = list[i];
    if (!o || o.d > 150) { vc.g.gain.setTargetAtTime(0, now, 0.05); continue; }
    const dx = o.x - me.x, dz = o.z - me.z, d = Math.max(o.d, 1);
    const vr = ((o.vx - me.vx) * dx + (o.vz - me.vz) * dz) / d;          // > 0 : elle s'éloigne
    const dop = clamp(190 / (190 + vr), 0.65, 1.5);
    const rpmO = gearInfo(o.sp).rpm, fo = (66 + rpmO * 660) * dop;
    const lat = -dx * sm + dz * cm;                                      // > 0 : à droite
    vc.a.frequency.setTargetAtTime(fo, now, 0.04); vc.b.frequency.setTargetAtTime(fo * 0.5, now, 0.04);
    vc.lp.frequency.setTargetAtTime(500 + rpmO * 2400 * clamp(1.15 - d / 160, 0.25, 1), now, 0.05);
    vc.g.gain.setTargetAtTime(0.19 * Math.pow(clamp(1 - d / 150, 0, 1), 1.7), now, 0.05);
    if (vc.pan) vc.pan.pan.setTargetAtTime(clamp(lat / 22, -1, 1), now, 0.05);
  }
}

let last = performance.now(), lastHud = 0, accT = 0;
function frame(t) {
  const dt = Math.min(0.05, Math.max(0.001, (t - last) / 1000)); last = t;

  if (phase === "countdown") {
    const lit = clamp(Math.floor((t - cdStart) / 600), 0, 5);
    lamps.forEach((l, i) => l.classList.toggle("on", i < lit));
    if (t >= raceStart) { setPhase("race"); flash("GO !", 900); }
  }

  // Voitures des autres : on prolonge leur trajectoire puis on lisse
  for (const id in remotes) {
    const r = remotes[id];
    if (t - r.tRecv > 4000) { delete remotes[id]; continue; }
    const ahead = Math.min(0.25, (t - r.tRecv) / 1000), sp = r.v || 0;
    const gx2 = r.x + Math.cos(r.a) * sp * ahead, gz2 = r.z + Math.sin(r.a) * sp * ahead;
    const kk = 1 - Math.exp(-12 * dt);
    r.rx += (gx2 - r.rx) * kk; r.rz += (gz2 - r.rz) * kk;
    const da = wrap(r.a - r.ra); r.ra += da * kk; r.sv = clamp(da * 2, -1, 1);
  }

  others = buildOthers();
  accT += dt; let n = 0;
  while (accT >= STEP && n < 10) { physics(STEP, t); accT -= STEP; n++; }
  if (n >= 10) accT = 0;

  if (phase === "countdown" || phase === "race" || phase === "finished") sendState(t);

  if (t - lastHud > 120) {
    lastHud = t;
    if (phase === "race" || phase === "finished" || phase === "countdown") { refreshTower(); refreshStats(t); }
    if (!$("results").hidden) refreshResults();
    refreshLobbyNote(t);
  }
  render3d(t, dt);
  drawGauge(dt); drawMini(); drawFx(); updateAudio(dt, t);
  requestAnimationFrame(frame);
}
