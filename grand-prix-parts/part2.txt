function physics(h, t) {
  const racing = phase === "race" || phase === "finished";
  if (!racing) return;
  const canDrive = phase === "race";
  const steerIn = canDrive ? ((input("right") ? 1 : 0) - (input("left") ? 1 : 0)) : 0;
  let thr = canDrive && input("up") ? 1 : 0, brk = canDrive && input("down") ? 1 : 0;
  if(canDrive&&!me.pitMode){
  const rel=pitRelativeFromCar(me),nr=nearest(me.x,me.z,me.idx);
  const right=((me.x-TX[nr.i])*RGX[nr.i]+(me.z-TZ[nr.i])*RGZ[nr.i])>TRACK_W*.16;
  if(rel>PIT_TRIG_REL-8&&rel<PIT_TRIG_REL+8&&right&&(steerIn>.05||me.speed<55)) startPlayerPit(rel);
}
  if (phase === "finished") { thr = 0; brk = 0.35; }
  me.thr = thr; me.brk = brk;
  if (canDrive && !me.dq) updateTires(me, t);
  if(canDrive&&me.pitMode)stepPlayerPit(h); else stepCar(me,thr,brk,steerIn,h);
  if(me.pitCrossFlag){me.pitCrossFlag=0;if(phase==="race")onLap(t);}   // ligne franchie dans la voie des stands = tour compté
  if(!me.pitMode){if(separate(me,others))sfxBump(Math.abs(me.speed));const r=locate(me);if(r.hit){camS.shake=1;sfxCrash(clamp(Math.abs(me.speed)/55,.25,1));sparkBurst(me.x,me.z,16);}if(r.crossed&&phase==="race")onLap(t);}

  if (isHost && phase !== "countdown" && bots.length) {
    const all = others.concat([me]);
    for (const b of bots) {
      const c = botControl(b, all, botSkill(b));

      b.thr = c.thr; b.brk = c.brk;
      if(!b.pitMode&&b.tireLaps>=2){const rel=pitRelativeFromCar(b);if(rel>PIT_TRIG_REL-9&&rel<PIT_TRIG_REL+9){b.pitMode=1;pitBegin(b,b.pitSlot||0,28);b.pitService=0;}}
      if(b.pitMode===1){if(pitMove(b,h,32)==="service")b.pitMode=2;if(b.pitCrossFlag){b.pitCrossFlag=0;b.tireLaps++;if(b.cross===laps+1)b.f=raceNow(t);}}
      else if(b.pitMode===2){const pp=pitBoxFrame(b.pitBox);b.x=pp.x;b.z=pp.z;b.a=pp.a;b.speed=0;b.vx=b.vz=0;b.pitService+=h;if(b.pitService>=PIT_STOP_DUR){b.tireLaps=b.pitSawLine?1:0;b.pitCompleted=0;b.pitDeadline=0;b.pitMode=3;pitLeave(b);}}
      else if(b.pitMode===3){if(pitMove(b,h,32)==="done")b.pitMode=0;if(b.pitCrossFlag){b.pitCrossFlag=0;b.tireLaps++;if(b.cross===laps+1)b.f=raceNow(t);}}
      else stepCar(b,c.thr,c.brk,c.steer,h);
      if(!b.pitMode){
        separate(b,all);
      }
      const q=b.pitMode ? {hit:false,crossed:0} : locate(b);
      if(q.crossed)b.tireLaps++;
      if(q.crossed&&b.cross===laps+1)b.f=raceNow(t);
      if (!b.f && !b.pitMode && Math.abs(b.speed) < 2) b.stuck += h; else b.stuck = 0;
      if (b.stuck > 3) {   // bot bloqué : on le remet sur la trajectoire
        b.x = TX[b.idx] + RGX[b.idx] * RLO[b.idx]; b.z = TZ[b.idx] + RGZ[b.idx] * RLO[b.idx];
        b.a = TA[b.idx]; b.vx = b.vz = b.speed = 0; b.stuck = 0;
      }
    }
  }
}

function standings() {
  const list = [{ id: me.id, me: true, name: me.name, color: me.color, p: me.progress, f: me.f, c: me.cross }];
  for (const id of raceOrder) {

    list.push({ id, me: false, name: nameOf(id), color: colorOf(id), p: r.p || 0, f: r.f || 0, c: r.c || 0 });
  }
  for (const id of raceBots) {
    if (isHost) {
      const b = bots.find(x => x.id === id);
      if (b) list.push({ id, me: false, name: nameOf(id), color: colorOf(id), p: b.progress, f: b.f, c: b.cross });

      const r = remotes[id];
      list.push({ id, me: false, name: nameOf(id), color: colorOf(id), p: r.p || 0, f: r.f || 0, c: r.c || 0 });
    }
  }
  list.sort((a, b) => (a.f && b.f) ? a.f - b.f : a.f ? -1 : b.f ? 1 : b.p - a.p);

// Ruban le long du circuit, entre deux décalages latéraux
function ribbon(inner, outer, y, material, tile) {
  const pos = [], uv = [], idx = [];
  const tiles = Math.max(1, Math.round(TRACK_LEN / tile)), vs = TRACK_LEN / tiles;
  let arc = 0;
  for (let i = 0; i <= N; i++) {
    const k = i % N, x = TX[k], z = TZ[k], rx = RGX[k], rz = RGZ[k];
    pos.push(x + rx * inner, y, z + rz * inner, x + rx * outer, y, z + rz * outer);
    uv.push(0, arc / vs, 1, arc / vs);
    if (i < N) { const j = (i + 1) % N; arc += Math.hypot(TX[j] - x, TZ[j] - z); }
  }
  for (let i = 0; i < N; i++) { const a = i * 2; idx.push(a, a + 1, a + 2, a + 1, a + 3, a + 2); }
  const geo = new THREE.BufferGeometry();
  geo.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
  geo.setAttribute("uv", new THREE.Float32BufferAttribute(uv, 2));
  geo.setIndex(idx);
  return new THREE.Mesh(geo, material);
}
// Mur vertical le long du circuit
function wallStrip(offset, height, material, tile, gapTest) {
  const pos = [], uv = [], idx = [];
  const tiles = Math.max(1, Math.round(TRACK_LEN / tile)), vs = TRACK_LEN / tiles;
  let arc = 0;
  // Important : une ouverture doit être une vraie ouverture.
  // L'ancienne version mettait simplement la hauteur à 0 sur certains sommets,
  // ce qui créait des triangles diagonaux et des "murs fantômes" dans la voie.
  for (let i = 0; i < N; i++) {
    const k = i, j = (i + 1) % N;
    const x0 = TX[k] + RGX[k] * offset, z0 = TZ[k] + RGZ[k] * offset;
    const x1 = TX[j] + RGX[j] * offset, z1 = TZ[j] + RGZ[j] * offset;
    const gap = gapTest && (gapTest(k) || gapTest(j));
    if (!gap) {
      const a = pos.length / 3;
      pos.push(x0, 0, z0, x0, height, z0, x1, 0, z1, x1, height, z1);
      uv.push(0, arc / vs, 1, arc / vs, 0, (arc + Math.hypot(x1-x0,z1-z0)) / vs, 1, (arc + Math.hypot(x1-x0,z1-z0)) / vs);
      idx.push(a, a + 1, a + 2, a + 1, a + 3, a + 2);
    }
    arc += Math.hypot(TX[j] - TX[k], TZ[j] - TZ[k]);
  }
  const geo = new THREE.BufferGeometry();
  geo.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
  geo.setAttribute("uv", new THREE.Float32BufferAttribute(uv, 2));
  geo.setIndex(idx);
  return new THREE.Mesh(geo, material);
}

function curveRadius(i, k) {
  const a = (i - k + N) % N, c = (i + k) % N;
  const abx = TX[i] - TX[a], abz = TZ[i] - TZ[a], acx = TX[c] - TX[a], acz = TZ[c] - TZ[a];
  const cross = Math.abs(abx * acz - abz * acx);
  const A = Math.hypot(abx, abz), B = Math.hypot(TX[c] - TX[i], TZ[c] - TZ[i]), C = Math.hypot(acx, acz);
  return cross > 1e-6 ? A * B * C / (2 * cross) : 1e9;
}
// Ruban dont la largeur suit la courbure (pour ne pas se replier dans les virages serrés)
function ribbonWide(side, y, material, tile, maxW) {
  const pos = [], uv = [], idx = [];
  const tiles = Math.max(1, Math.round(TRACK_LEN / tile)), vs = TRACK_LEN / tiles;
  let arc = 0;
  for (let i = 0; i <= N; i++) {
    const k = i % N, x = TX[k], z = TZ[k];
    let R = 1e9; for (let d = -3; d <= 3; d++) R = Math.min(R, curveRadius((k + d + N) % N, 6));
    const w = clamp(0.8 * R - WALL_D, 5, maxW), a = side * WALL_D, b = side * (WALL_D + w);
    pos.push(x + RGX[k] * a, y, z + RGZ[k] * a, x + RGX[k] * b, y, z + RGZ[k] * b);
    uv.push(0, arc / vs, 1, arc / vs);
    if (i < N) { const j = (i + 1) % N; arc += Math.hypot(TX[j] - x, TZ[j] - z); }
  }
  for (let i = 0; i < N; i++) { const a = i * 2; idx.push(a, a + 1, a + 2, a + 1, a + 3, a + 2); }
  const geo = new THREE.BufferGeometry();
  geo.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
  geo.setAttribute("uv", new THREE.Float32BufferAttribute(uv, 2));
  geo.setIndex(idx);
  return new THREE.Mesh(geo, material);
}
// Bande verticale avec une image qui se lit dans le sens de la piste (panneaux publicitaires)
function wallStripUV(offset, y0, y1, material, tile, flip, gapTest) {
  const pos = [], uv = [], idx = [];
  const tiles = Math.max(1, Math.round(TRACK_LEN / tile)), vs = TRACK_LEN / tiles;
  let arc = 0;
  // Comme wallStrip : là où gapTest est vrai (entrée / sortie des stands) il n'y a PAS de panneau,
  // sinon la voiture traversait les panneaux publicitaires en entrant et en sortant des stands.
  for (let i = 0; i < N; i++) {
    const k = i, j = (i + 1) % N, seg = Math.hypot(TX[j] - TX[k], TZ[j] - TZ[k]);
    if (!(gapTest && (gapTest(k) || gapTest(j)))) {
      const a = pos.length / 3, u0 = (flip ? -1 : 1) * arc / vs, u1 = (flip ? -1 : 1) * (arc + seg) / vs;
      pos.push(TX[k] + RGX[k] * offset, y0, TZ[k] + RGZ[k] * offset, TX[k] + RGX[k] * offset, y1, TZ[k] + RGZ[k] * offset,
               TX[j] + RGX[j] * offset, y0, TZ[j] + RGZ[j] * offset, TX[j] + RGX[j] * offset, y1, TZ[j] + RGZ[j] * offset);
      uv.push(u0, 0, u0, 1, u1, 0, u1, 1);
      idx.push(a, a + 1, a + 2, a + 1, a + 3, a + 2);
    }
    arc += seg;
  }
  const geo = new THREE.BufferGeometry();
  geo.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
  geo.setAttribute("uv", new THREE.Float32BufferAttribute(uv, 2));
  geo.setIndex(idx);
  return new THREE.Mesh(geo, material);
}

const PVERT = "attribute float aSize;attribute float aAlpha;attribute vec3 aColor;uniform float uScale;varying float vA;varying vec3 vC;" +
  "void main(){vec4 mv=modelViewMatrix*vec4(position,1.0);gl_Position=projectionMatrix*mv;" +
  "gl_PointSize=min(aSize*uScale/max(-mv.z,0.5),180.0);vA=aAlpha;vC=aColor;}";
const PFRAG = "varying float vA;varying vec3 vC;" +
  "void main(){float r=length(gl_PointCoord-vec2(0.5))*2.0;float a=smoothstep(1.0,0.15,r)*vA;if(a<0.01)discard;gl_FragColor=vec4(vC,a);}";
function makeParticles(max, additive) {
  const geo = new THREE.BufferGeometry();
  const pos = new Float32Array(max * 3), col = new Float32Array(max * 3), siz = new Float32Array(max), alp = new Float32Array(max);
  geo.setAttribute("position", new THREE.BufferAttribute(pos, 3));
  geo.setAttribute("aColor", new THREE.BufferAttribute(col, 3));
  geo.setAttribute("aSize", new THREE.BufferAttribute(siz, 1));
  geo.setAttribute("aAlpha", new THREE.BufferAttribute(alp, 1));
  const mat = new THREE.ShaderMaterial({
    uniforms: { uScale: { value: 700 } }, vertexShader: PVERT, fragmentShader: PFRAG,
    transparent: true, depthWrite: false, blending: additive ? THREE.AdditiveBlending : THREE.NormalBlending
  });
  const pts = new THREE.Points(geo, mat); pts.frustumCulled = false;
  const vx = new Float32Array(max), vy = new Float32Array(max), vz = new Float32Array(max);
  const age = new Float32Array(max), life = new Float32Array(max), s0 = new Float32Array(max), s1 = new Float32Array(max);
  const a0 = new Float32Array(max), grav = new Float32Array(max), drag = new Float32Array(max);
  age.fill(1); life.fill(1);
  let cur = 0, dirty = false;
  function emit(x, y, z, ax, ay, az, lf, sa, sb, r, g, b, al, gr, dr) {
    const i = cur; cur = (cur + 1) % max;
    pos[i * 3] = x; pos[i * 3 + 1] = y; pos[i * 3 + 2] = z; vx[i] = ax; vy[i] = ay; vz[i] = az;
    age[i] = 0; life[i] = lf; s0[i] = sa; s1[i] = sb; a0[i] = al; grav[i] = gr; drag[i] = dr;
    col[i * 3] = r; col[i * 3 + 1] = g; col[i * 3 + 2] = b; siz[i] = sa; alp[i] = al; dirty = true;
  }
  function update(dt) {
    if (!dirty) return;
    let alive = 0;
    for (let i = 0; i < max; i++) {
      if (age[i] >= life[i]) { if (alp[i] !== 0) { alp[i] = 0; siz[i] = 0; } continue; }
      age[i] += dt; const k = age[i] / life[i];
      if (k >= 1) { alp[i] = 0; siz[i] = 0; continue; }
      alive++;
      const dd = Math.exp(-drag[i] * dt);
      vx[i] *= dd; vz[i] *= dd; vy[i] = vy[i] * dd + grav[i] * dt;
      pos[i * 3] += vx[i] * dt; pos[i * 3 + 1] += vy[i] * dt; pos[i * 3 + 2] += vz[i] * dt;
      if (pos[i * 3 + 1] < 0.05 && grav[i] < 0) { pos[i * 3 + 1] = 0.05; vy[i] *= -0.3; }
      siz[i] = s0[i] + (s1[i] - s0[i]) * k;
      alp[i] = a0[i] * Math.min(1, (1 - k) * 1.6);
    }
    geo.attributes.position.needsUpdate = true; geo.attributes.aColor.needsUpdate = true;
    geo.attributes.aSize.needsUpdate = true; geo.attributes.aAlpha.needsUpdate = true;
    if (!alive) dirty = false;
  }
  return { pts, mat, emit, update };
}
const pickCount = n => { const c = Math.floor(n); return c + (Math.random() < n - c ? 1 : 0); };
// Dérapage d'une voiture : 0 = adhère, 1 = glisse à fond
function slipOf(c) {
  const v = Math.abs(c.speed); if (v < 10) return 0;
  const vl = -c.vx * Math.sin(c.a) + c.vz * Math.cos(c.a);          // vitesse sur le côté
  let s = clamp((Math.abs(vl) - 1.8) / 2.6, 0, 1);
  if (c.brk > 0.8 && v > 30) s = Math.max(s, clamp((v - 30) / 55, 0, 0.7));
  if (c.dist > OFFD) s *= 0.4;
  return s;
}

  if (!sparks) return;

function stepFx(dt) {
  if (!smoke || !sparks) return;
  fxPop = Math.max(0, fxPop - dt * 6);
  if (phase === "race" || phase === "finished") {
    const ca = Math.cos(me.a), sa = Math.sin(me.a), v = Math.abs(me.speed);
    const rear = () => { const sd = Math.random() < 0.5 ? -1 : 1, lx = -1.8, lz = sd * 1.05; return [me.x + ca * lx - sa * lz, me.z + sa * lx + ca * lz]; };
    const slip = slipOf(me);
    if (slip > 0.05) {                                              // fumée blanche des pneus
      for (let k = pickCount(slip * 80 * dt); k > 0; k--) {
        const p = rear();
        smoke.emit(p[0], 0.25, p[1], me.vx * 0.25 + (Math.random() - 0.5) * 1.6, 0.8 + Math.random() * 1.2, me.vz * 0.25 + (Math.random() - 0.5) * 1.6,
          1.0 + Math.random() * 0.7, 1.4, 5.5, 0.93, 0.93, 0.95, 0.12 + 0.4 * slip, 0, 1.2);
      }
    }
    if (me.dist > OFFD && v > 5) {                                  // poussière et gravier hors piste
      for (let k = pickCount(Math.min(1, v / 40) * 70 * dt); k > 0; k--) {
        const p = rear(), c = Math.random() < 0.55 ? [0.74, 0.64, 0.44] : [0.38, 0.56, 0.3];
        smoke.emit(p[0], 0.2, p[1], me.vx * 0.2 + (Math.random() - 0.5) * 3, 1.5 + Math.random() * 2.5, me.vz * 0.2 + (Math.random() - 0.5) * 3,
          0.8 + Math.random() * 0.7, 0.9, 3.4, c[0], c[1], c[2], 0.5, -2, 1.5);
      }
    }
  }
  smoke.update(dt); sparks.update(dt);
  const sc = (H * DPR * 0.5) / Math.tan(camera.fov * Math.PI / 360);
  smoke.mat.uniforms.uScale.value = sc; sparks.mat.uniforms.uScale.value = sc;
}

function bakeParts(parts) {
  const d = new THREE.Object3D();
  const list = parts.map(p => {
    const g = p.g.clone();
    d.position.set(p.x || 0, p.y || 0, p.z || 0);
    d.rotation.order = p.o || "XYZ"; d.rotation.set(p.rx || 0, p.ry || 0, p.rz || 0);
    d.scale.set(p.sx || 1, p.sy || 1, p.sz || 1);
    d.updateMatrix(); g.applyMatrix4(d.matrix);
    return { g, c: new THREE.Color(p.c) };
  });
  let vc = 0, ic = 0;
  list.forEach(e => { vc += e.g.attributes.position.count; ic += e.g.index.count; });
  const pos = new Float32Array(vc * 3), nor = new Float32Array(vc * 3), col = new Float32Array(vc * 3);
  const idx = new (vc > 65000 ? Uint32Array : Uint16Array)(ic);
  let vo = 0, io = 0;
  list.forEach(e => {
    const P = e.g.attributes.position, Nn = e.g.attributes.normal, I = e.g.index;
    pos.set(P.array, vo * 3); nor.set(Nn.array, vo * 3);
    for (let i = 0; i < P.count; i++) { col[(vo + i) * 3] = e.c.r; col[(vo + i) * 3 + 1] = e.c.g; col[(vo + i) * 3 + 2] = e.c.b; }
    for (let i = 0; i < I.count; i++) idx[io + i] = I.array[i] + vo;
    vo += P.count; io += I.count;
  });
  const geo = new THREE.BufferGeometry();
  geo.setAttribute("position", new THREE.BufferAttribute(pos, 3));
  geo.setAttribute("normal", new THREE.BufferAttribute(nor, 3));
  geo.setAttribute("color", new THREE.BufferAttribute(col, 3));
  geo.setIndex(new THREE.BufferAttribute(idx, 1));
  geo.computeBoundingSphere();
  return geo;
}
